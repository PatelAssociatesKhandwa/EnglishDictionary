/* ===================================================== RESULT ===================================================== */ function showExamResult( correct, incorrect, attempted, unanswered, total, percentage, elapsedSeconds ){ document.getElementById( "examArea" ).classList.add( "hidden" ); const result = document.getElementById( "examResult" ); result.classList.remove( "hidden" ); result.innerHTML = `

Exam Result

${correct} / ${total}
Total Questions ${total}
Attempted ${attempted}
Correct ${correct}
Incorrect ${incorrect}
Unattempted ${unanswered}
Time Taken
${formatTime( elapsedSeconds )}
Accuracy
${percentage}%
`; /* ========================================= PIE CHART ONLY ========================================== */ if( typeof Chart !== "undefined" ){ const pieCanvas = document.getElementById( "examPieChart" ); if( typeof ChartDataLabels !== "undefined" ){ Chart.register( ChartDataLabels ); } new Chart( pieCanvas, { type: "pie", data:{ labels:[ "Correct", "Incorrect", "Unattempted" ], datasets:[ { data:[ correct, incorrect, unanswered ] } ] }, options:{ responsive: true, maintainAspectRatio: false, plugins:{ legend:{ display: false }, title:{ display: true, text: "Performance", font:{ size:14 } }, datalabels:{ display: true, formatter: function( value, context ){ const totalValue = context .chart .data .datasets[0] .data .reduce( function( a, b ){ return a + b; }, 0 ); if( !totalValue ){ return "0%"; } return Math.round( ( value / totalValue ) * 100 ) + "%"; }, font:{ weight: "bold", size: 12 } } } } } ); } }
